struct Count
  lo: Int
  hi: Int

  fun next(!self) -> Int?
    return none if self.lo >= self.hi
    self.lo += 1
    self.lo - 1

fun upto(n: Int) -> Count
  Count(lo: 0, hi: n)

struct Tracked
  id: Int
  left: Int

  fun next(!self) -> Int?
    return none if self.left == 0
    self.left -= 1
    self.id * 10 + self.left

  drop(!self)
    print("drop", self.id)

fun tracked(id: Int, n: Int) -> Tracked
  Tracked(id: id, left: n)

struct Item
  name: String
  n: Int

struct Walk
  items: ?Vec[Item]
  at: Int

  fun next(!self) -> (?Item)?
    return none if self.at >= self.items.len
    i = self.at
    self.at += 1
    ?self.items[i]

fun walk(items: ?Vec[Item]) -> Walk from items
  Walk(items: items, at: 0)

fun mkitems() -> Vec[Item]
  v: Vec[Item] = Vec()
  !v.push(Item(name: "a", n: 1))
  !v.push(Item(name: "b", n: 2))
  v
struct Buf
  buf: Vec[Item]
  at: Int
  fun next(!self) -> (?Item)?
    return none if self.at >= self.buf.len
    self.at += 1
    ?self.buf[self.at - 1]
fun mkbuf() -> Buf
  Buf(buf: mkitems(), at: 0)
